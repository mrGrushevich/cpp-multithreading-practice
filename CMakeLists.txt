cmake_minimum_required(VERSION 3.16)
project(cpp_practice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
foreach(name IN ITEMS 01_threads 02_race 03_mutex_atomic 04_async 05_stop_token 06_platform generate_logs logscan)
    add_executable(${name} src/${name}.cpp)
    target_link_libraries(${name} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
# Makes lost updates easier to observe; it does NOT make the program well-defined.
if(MSVC)
    target_compile_options(02_race PRIVATE /Od)
else()
    target_compile_options(02_race PRIVATE -O0)
endif()
